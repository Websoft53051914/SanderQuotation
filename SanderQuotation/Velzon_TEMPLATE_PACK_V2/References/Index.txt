        function escapeHtml(text) {
            return String(text || '')
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#39;');
        }

        function getStatusBadge(status) {
            if (status === '已生效') return 'bg-success-subtle text-success';
            if (status === '簽核中') return 'bg-warning-subtle text-warning';
            return 'bg-secondary-subtle text-secondary';
        }

        function collectFilteredRows() {
            const k1 = queryState.keyword1.toLowerCase();
            const k2 = queryState.keyword2.toLowerCase();
            const g = queryState.globalSearch.toLowerCase();

            let rows = dataRows.filter(function (row) {
                const source = [row.no, row.name, row.type, row.owner, row.status, row.updatedAt].join(' ').toLowerCase();
                const hitK1 = !k1 || row.name.toLowerCase().includes(k1) || row.type.toLowerCase().includes(k1);
                const hitK2 = !k2 || row.owner.toLowerCase().includes(k2);
                const hitG = !g || source.includes(g);
                const hitStatus = !queryState.status || row.status === queryState.status;
                return hitK1 && hitK2 && hitG && hitStatus;
            });

            return rows;
        }

        function renderTable() {
            const allRows = collectFilteredRows();
            const tbody = $(DOM.tbody);
            tbody.empty();

            const total = allRows.length;
            const start = (queryState.page - 1) * queryState.pageSize;
            const pageRows = allRows.slice(start, start + queryState.pageSize);

            if (pageRows.length === 0) {
                $(DOM.noResult).show();
            } else {
                $(DOM.noResult).hide();
            }

            pageRows.forEach(function (row) {
                tbody.append(

                    '<td>' + escapeHtml(row.updatedAt) + '</td>' +
                    '</tr>'
                );
            });

            $(DOM.removeActions).hide();
            $(DOM.checkAll).prop('checked', false);
            renderPagination(total);
        }

        function renderPagination(total) {
            const totalPages = Math.max(1, Math.ceil(total / queryState.pageSize));
            if (queryState.page > totalPages) queryState.page = totalPages;

            const ul = $(DOM.pagination);
            ul.empty();

            const maxPages = 5;
            let startPage = Math.max(1, queryState.page - Math.floor(maxPages / 2));
            let endPage = Math.min(totalPages, startPage + maxPages - 1);
            startPage = Math.max(1, endPage - maxPages + 1);

            for (let i = startPage; i <= endPage; i++) {
                ul.append('<li class="page-item ' + (i === queryState.page ? 'active' : '') + '"><a class="page-link" href="javascript:void(0)" onclick="gotoPage(' + i + ')">' + i + '</a></li>');
            }

            $(DOM.pagePrev).toggleClass('disabled', queryState.page <= 1);
            $(DOM.pageNext).toggleClass('disabled', queryState.page >= totalPages);
        }

        function updateRemoveActions() {
            const checkedCount = $(DOM.rowCheck + ':checked').length;
            if (checkedCount > 0) {
                $(DOM.removeActions).show();
            } else {
                $(DOM.removeActions).hide();
            }
        }

        function initChoices(selector) {
            $(selector).each(function () {
                if (this.choices || typeof Choices === 'undefined') return;
                this.choices = new Choices(this, {
                    searchEnabled: this.attributes.searchenabled != null,
                    itemSelectText: '',
                    removeItemButton: false
                });
            });
        }

        function setSelectValue($el, value) {
            if ($el[0] && $el[0].choices) {
                $el[0].choices.removeActiveItems();
                if (value !== null && value !== undefined) {
                    $el[0].choices.setChoiceByValue(String(value));
                }
            } else {
                $el.val(value);
            }
        }

        function resetForm() {
            const form = $(DOM.submitForm)[0];
            form.reset();
            $(DOM.submitForm).removeClass('was-validated');
            $('#ApplicableRolesGroup').removeClass('is-invalid');
            $('#RowId').val('');
            pendingFileName = '';
            setSelectValue($('#SopType'), '');

            if (window.flatpickr) {
                const startPicker = document.querySelector('#StartDate')._flatpickr;
                const endPicker = document.querySelector('#EndDate')._flatpickr;
                if (startPicker) startPicker.clear();
                if (endPicker) endPicker.clear();
            }
        }

        function getSelectedRoles() {
            return $('.role-item:checked').map(function () { return this.value; }).get();
        }

        function validateRoleGroup() {
            const roles = getSelectedRoles();
            const isValid = roles.length > 0;
            $('#ApplicableRolesGroup').toggleClass('is-invalid', !isValid);
            return isValid;
        }

        function validateDateRange() {
            const startVal = $('#StartDate').val();
            const endVal = $('#EndDate').val();
            if (!startVal || !endVal) return true;

            const start = new Date(startVal.replace(' ', 'T'));
            const end = new Date(endVal.replace(' ', 'T'));
            if (Number.isNaN(start.getTime()) || Number.isNaN(end.getTime())) return true;

            if (end <= start) {
                Swal.fire({ icon: 'warning', title: '結束日期必須晚於開始日期' });
                return false;
            }
            return true;
        }

            $('#Version').val(row.version);
            setSelectValue($('#SopType'), row.type);
            $('#Owner').val(row.owner);
            $('input[name="RiskLevel"]').prop('checked', false);
            $('input[name="RiskLevel"][value="' + row.risk + '"]').prop('checked', true);
            $('#Description').val(row.description || '');

            $('.role-item').prop('checked', false);
            (row.roles || []).forEach(function (value) {
                $('.role-item[value="' + value + '"]').prop('checked', true);

            if (window.flatpickr) {
                const startPicker = document.querySelector('#StartDate')._flatpickr;
                const endPicker = document.querySelector('#EndDate')._flatpickr;
                if (startPicker) startPicker.setDate(row.startDate, true, 'Y-m-d H:i');
                if (endPicker) endPicker.setDate(row.endDate, true, 'Y-m-d H:i');
            } else {
                $('#StartDate').val(row.startDate);
                $('#EndDate').val(row.endDate);
            }
        }

        function saveForm() {
            const form = $(DOM.submitForm)[0];
            const rolesValid = validateRoleGroup();
            const datesValid = validateDateRange();
            if (!form.checkValidity() || !rolesValid || !datesValid) {
                $(DOM.submitForm).addClass('was-validated');
                return;
            }

            const nowText = new Date().toISOString().slice(0, 16).replace('T', ' ');
            const rowId = $('#RowId').val();
            const riskValue = $('input[name="RiskLevel"]:checked').val() || '低';
            const oldRow = rowId ? dataRows.find(function (x) { return x.id === rowId; }) : null;
            const nextStatus = oldRow ? oldRow.status : '草稿';

            const payload = {
                id: rowId || ('s' + Date.now()),
                no: rowId ? (oldRow || {}).no : ('SOP-' + String(dataRows.length + 1).padStart(3, '0')),
                name: $('#SopName').val().trim(),
                type: ($('#SopType')[0] && $('#SopType')[0].choices) ? $('#SopType')[0].choices.getValue(true) : $('#SopType').val(),
                version: Number($('#Version').val()),
                startDate: $('#StartDate').val(),
                endDate: $('#EndDate').val(),
                status: nextStatus,
                owner: $('#Owner').val().trim() || '未指定',
                updatedAt: nowText,
                isEnabled: nextStatus === '已生效',
                roles: getSelectedRoles(),
                risk: riskValue,
                description: $('#Description').val().trim(),
                files: pendingFileName ? [pendingFileName] : []
            };

            if (rowId) {
                dataRows = dataRows.map(function (item) {
                    return item.id === rowId ? payload : item;
                });
            } else {
                dataRows.unshift(payload);
            }

            const modal = bootstrap.Modal.getInstance(document.querySelector(DOM.showModal));
            if (modal) modal.hide();
            renderTable();
            Swal.fire({ icon: 'success', title: '儲存成功', timer: 1200, showConfirmButton: false });
        }

        function deleteByIds(ids) {
            if (!ids || ids.length === 0) return;
            dataRows = dataRows.filter(function (row) { return ids.indexOf(row.id) === -1; });
            renderTable();
            Swal.fire({ icon: 'success', title: '刪除完成', timer: 1000, showConfirmButton: false });
        }

        function initDatepickers() {
            if (window.flatpickr) {
                const opts = {
                    enableTime: true,
                    dateFormat: 'Y-m-d H:i',
                    time_24hr: true
                };
                flatpickr('#StartDate', opts);
                flatpickr('#EndDate', opts);
            } else {
                $('#StartDate').attr('type', 'datetime-local');
                $('#EndDate').attr('type', 'datetime-local');
            }
        }

        function gotoPage(page) {
            queryState.page = page;
            renderTable();
        }

        function SearchData() {
            queryState.keyword1 = $('#Keyword1').val().trim();
            queryState.keyword2 = $('#Keyword2').val().trim();
            queryState.status = ($('#StatusFilter')[0] && $('#StatusFilter')[0].choices) ? $('#StatusFilter')[0].choices.getValue(true) : $('#StatusFilter').val();
            queryState.page = 1;
            renderTable();
        }

        function ClearSearch() {
            $('#globalSearchInput').val('');
            $('#Keyword1').val('');
            $('#Keyword2').val('');
            setSelectValue($('#StatusFilter'), '');
            queryState.globalSearch = '';
            queryState.keyword1 = '';
            queryState.keyword2 = '';
            queryState.status = '';
            queryState.page = 1;
            renderTable();
        }

        function deleteMultiple() {
            const ids = $(DOM.rowCheck + ':checked').map(function () { return $(this).data('id'); }).get();
            if (ids.length === 0) {
                Swal.fire({ icon: 'warning', title: '請至少勾選一筆資料' });
                return;
            }

            Swal.fire({
                icon: 'question',
                title: '確定要刪除選取的 ' + ids.length + ' 筆資料嗎？',
                showCancelButton: true,
                confirmButtonText: '確定',
                cancelButtonText: '取消'
            }).then(function (result) {
                if (result.isConfirmed) deleteByIds(ids);
            });
        }

        window.gotoPage = gotoPage;
        window.SearchData = SearchData;
        window.ClearSearch = ClearSearch;
        window.deleteMultiple = deleteMultiple;

        $(document).ready(function () {
            initDatepickers();
            initChoices('#StatusFilter, #SopType');
            renderTable();

            $(DOM.pagePrev).on('click', function (e) {
                e.preventDefault();
                if (queryState.page > 1) {
                    queryState.page -= 1;
                    renderTable();
                }
            });

            $(DOM.pageNext).on('click', function (e) {
                e.preventDefault();
                const totalPages = Math.max(1, Math.ceil(collectFilteredRows().length / queryState.pageSize));
                if (queryState.page < totalPages) {
                    queryState.page += 1;
                    renderTable();
                }
            });

            $(DOM.globalSearch).on('input', function () {
                queryState.globalSearch = $(this).val().trim();
                queryState.page = 1;
                renderTable();
            });

            $(DOM.checkAll).on('change', function () {
                $(DOM.rowCheck).prop('checked', this.checked);
                updateRemoveActions();
            });

            $(document).on('change', DOM.rowCheck, function () {
                const total = $(DOM.rowCheck).length;
                const checked = $(DOM.rowCheck + ':checked').length;
                $(DOM.checkAll).prop('checked', total > 0 && total === checked);
                updateRemoveActions();
            });

            $(DOM.btnShowModal).on('click', function () {
                currentId = null;
                $(DOM.labelCreate).show();
                $(DOM.labelEdit).hide();
                resetForm();
            });

            $(document).on('click', '.edit-item-btn', function () {
                const id = $(this).closest('tr').data('id');
                const row = dataRows.find(function (item) { return item.id === id; });
                if (!row) return;
                currentId = id;
                $(DOM.labelCreate).hide();
                $(DOM.labelEdit).show();
                resetForm();
                fillForm(row);
            });

            $(document).on('click', '.remove-item-btn', function () {
                currentId = $(this).closest('tr').data('id');
            });

            $(DOM.deleteRecord).on('click', function () {
                if (!currentId) return;
                const modal = bootstrap.Modal.getInstance(document.querySelector(DOM.deleteModal));
                if (modal) modal.hide();
                deleteByIds([currentId]);
                currentId = null;
            });

            $(DOM.submitForm).on('submit', function (e) {
                e.preventDefault();
                saveForm();
            });

            $('.role-item').on('change', function () {
                validateRoleGroup();
            });

            $('#demoFileInput').on('change', function () {
                const file = (this.files && this.files.length > 0) ? this.files[0] : null;
                pendingFileName = file ? file.name : '';
            });
        });
    </script>
}
